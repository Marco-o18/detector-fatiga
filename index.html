async function enviarComando(valorString) { // 1. Si no hay una conexión Bluetooth activa, salimos silenciosamente sin congelar nada if (!bleCharacteristic) { console.log("Intento de envío sin conexión Bluetooth activa: ", valorString); return; } try { const encoder = new TextEncoder(); // 2. Intentamos enviar el dato de forma asíncrona await bleCharacteristic.writeValue(encoder.encode(valorString)); console.log("Comando enviado con éxito: ", valorString); } catch (err) { // 3. Si la tablet se desconectó o falló el envío por interferencia, // atrapamos el error aquí para que la cámara SIGA FUNCIONANDO normalmente. console.error("Error al enviar dato por Bluetooth, limpiando conexión: ", err); // Reseteamos las variables de estado para obligar a reconectar si falla bleCharacteristic = null; const btn = document.getElementById('btn-bluetooth'); btn.innerText = "Conexión Perdida (Reconectar)"; btn.classList.remove("conectado"); } }
